```swift
import EventKit
import SwiftUI

struct CalendarExportResult {
    let created: Int
    let updated: Int
    let outcome: CalendarExportOutcome
}

enum CalendarExportOutcome {
    case success
    case permissionDenied
    case permissionRestricted
    case externallyDeletedEventsDetected(count: Int)
}

final class CalendarExportService {

    private let eventStore = EKEventStore()

    func exportDay(_ day: Day, context: ModelContext) async throws -> CalendarExportResult {
        guard await hasPermission() else {
            return CalendarExportResult(created: 0, updated: 0, outcome: .permissionDenied)
        }

        var createdCount = 0
        var updatedCount = 0

        for stop in day.stops where stop.name.isNotEmpty && (stop.startTime != nil || stop.endTime != nil) {
            if let identifier = stop.calendarEventIdentifier, identifier.isNotEmpty {
                do {
                    try await updateEvent(withIdentifier: identifier, day: day)
                    updatedCount += 1
                } catch {
                    // Handle specific errors like event not found
                    continue
                }
            } else {
                try await createNewEvent(for: stop, day: day)
                createdCount += 1
            }
        }

        return CalendarExportResult(created: createdCount, updated: updatedCount, outcome: .success)
    }

    private func hasPermission() async -> Bool {
        switch await eventStore.requestFullAccessToEvents() {
        case .denied, .restricted:
            return false
        case .authorized, .undetermined:
            return true
        @unknown default:
            return false
        }
    }

    private func createNewEvent(for stop: Stop, day: Day) async throws {
        let event = EKEvent(eventStore: eventStore)
        event.title = stop.name
        event.startDate = stop.startTime ?? day.date
        event.endDate = stop.endTime ?? Calendar.current.date(byAdding: .hour, value: 1, to: day.date)!
        event.calendar = eventStore.defaultCalendarForNewEvents

        try await saveEvent(event, for: stop)
    }

    private func updateEvent(withIdentifier identifier: String, day: Day) async throws {
        if let event = eventStore.event(withIdentifier: identifier) {
            event.startDate = day.date
            event.endDate = Calendar.current.date(byAdding: .hour, value: 1, to: day.date)!
            try await saveEvent(event)
        } else {
            throw NSError(domain: "Waypoint", code: -1, userInfo: [NSLocalizedDescriptionKey: "Event not found"])
        }
    }

    private func saveEvent(_ event: EKEvent) async throws {
        let span = Calendar.current.dateComponents([.year, .month, .day], from: event.startDate)
        if !eventStore.supportsAvailability(forEventsIn: span) {
            throw NSError(domain: "Waypoint", code: -2, userInfo: [NSLocalizedDescriptionKey: "Event store does not support availability"])
        }

        try await eventStore.save(event, span: .thisEvent)
    }
}
```

### Explanation:

1. **CalendarExportResult and CalendarExportOutcome**
   - These structs represent the result of exporting a day’s stops, including counts of created/updated events and the outcome status.

2. **exportDay(_:context:)**
   - This method checks for permissions before proceeding.
   - It iterates over each stop in the day, either creating or updating calendar events based on whether an `calendarEventIdentifier` is present.
   
3. **hasPermission()**
   - Requests full access to events and returns a Boolean indicating if permission was granted.

4. **createNewEvent(for:day:)**
   - Creates a new event in the default calendar with the stop's name, start/end times derived from the day’s date.
   - Saves the event using `saveEvent(_:)`.

5. **updateEvent(withIdentifier:day:)**
   - Updates an existing event identified by `calendarEventIdentifier` with the day’s date and end time.
   - Saves the updated event using `saveEvent(_:)`.

6. **saveEvent(_:)**
   - Validates that the event store supports availability for the given span before saving the event.

### Implementing `CalendarExportService`

**File: WaypointKit/Sources/WaypointKit/Services/CalendarExportService.swift**

```swift
import EventKit
import SwiftUI

struct CalendarExportResult {
    let created: Int
    let updated: Int
    let outcome: CalendarExportOutcome
}

enum CalendarExportOutcome {
    case success
    case permissionDenied
    case permissionRestricted
    case externallyDeletedEventsDetected(count: Int)
}

final class CalendarExportService {

    private let eventStore = EKEventStore()

    func exportDay(_ day: Day, context: ModelContext) async throws -> CalendarExportResult {
        guard await hasPermission() else {
            return CalendarExportResult(created: 0, updated: 0, outcome: .permissionDenied)
        }

        var createdCount = 0
        var updatedCount = 0

        for stop in day.stops where stop.name.isNotEmpty && (stop.startTime != nil || stop.endTime != nil) {
            if let identifier = stop.calendarEventIdentifier, identifier.isNotEmpty {
                do {
                    try await updateEvent(withIdentifier: identifier, day: day)
                    updatedCount += 1
                } catch {
                    // Handle specific errors like event not found
                    continue
                }
            } else {
                try await createNewEvent(for: stop, day: day)
                createdCount += 1
            }
        }

        return CalendarExportResult(created: createdCount, updated: updatedCount, outcome: .success)
    }

    private func hasPermission() async -> Bool {
        switch await eventStore.requestFullAccessToEvents() {
        case .denied, .restricted:
            return false
        case .authorized, .undetermined:
            return true
        @unknown default:
            return false
        }
    }

    private func createNewEvent(for stop: Stop, day: Day) async throws {
        let event = EKEvent(eventStore: eventStore)
        event.title = stop.name
        event.startDate = stop.startTime ?? day.date
        event.endDate = stop.endTime ?? Calendar.current.date(byAdding: .hour, value: 1, to: day.date)!
        event.calendar = eventStore.defaultCalendarForNewEvents

        try await saveEvent(event, for: stop)
    }

    private func updateEvent(withIdentifier identifier: String, day: Day) async throws {
        if let event = eventStore.event(withIdentifier: identifier) {
            event.startDate = day.date
            event.endDate = Calendar.current.date(byAdding: .hour, value: 1, to: day.date)!
            try await saveEvent(event)
        } else {
            throw NSError(domain: "Waypoint", code: -1, userInfo: [NSLocalizedDescriptionKey: "Event not found"])
        }
    }

    private func saveEvent(_ event: EKEvent) async throws {
        let span = Calendar.current.dateComponents([.year, .month, .day], from: event.startDate)
        if !eventStore.supportsAvailability(forEventsIn: span) {
            throw NSError(domain: "Waypoint", code: -2, userInfo: [NSLocalizedDescriptionKey: "Event store does not support availability"])
        }

        try await eventStore.save(event, span: .thisEvent)
    }
}
```

### Explanation:

1. **CalendarExportResult and CalendarExportOutcome**
   - These structs represent the result of exporting a day’s stops, including counts of created/updated events and the outcome status.

2. **exportDay(_:context:)**
   - This method checks for permissions before proceeding.
   - It iterates over each stop in the day, either creating or updating calendar events based on whether an `calendarEventIdentifier` is present.
   
3. **hasPermission()**
   - Requests full access to events and returns a Boolean indicating if permission was granted.

4. **createNewEvent(for:day:)**
   - Creates a new event in the default calendar with the stop's name, start/end times derived from the day’s date.
   - Saves the event using `saveEvent(_:)`.

5. **updateEvent(withIdentifier:day:)**
   - Updates an existing event identified by `calendarEventIdentifier` with the day’s date and end time.
   - Saves the updated event using `saveEvent(_:)`.

6. **saveEvent(_:)**
   - Validates that the event store supports availability for the given span before saving the event.
