enum DemoSeeder {

        if getDemoTrip(context: context) == nil {

    static func remove(context: ModelContext) {
        if let demoTrip = getDemoTrip(context: context) {
            context.delete(demoTrip)
        }
    }
    
    private static func getDemoTrip(context: ModelContext) -> Trip? {
        // Fetch the demo trip from your data store

            sortDescriptors: []

    }
    
    private static func createDemoTrip(context: ModelContext) {
