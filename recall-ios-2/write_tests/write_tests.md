```swift
import RecallCore
import UserNotifications

@Observable @MainActor final class NotificationScheduler {
    private(set) var authorizationStatus = UNAuthorizationStatus.notDetermined
    
    init() {
        refreshAuthorizationStatus()
    }
    
    func refreshAuthorizationStatus() async {
        let center = UNUserNotificationCenter.current()
        let (status, _) = try! await center.requestAuthorization(options: [.alert])
        authorizationStatus = status
    }
    
    func scheduleDailyReminder(at time: DateComponents) async throws {
        guard authorizationStatus == .authorized else { return }
        
        let center = UNUserNotificationCenter.current()
        let content = UNMutableNotificationContent()
        content.title = "Review Time"
        content.body = "It's time to review your flashcards!"
        
        var dateComponents = Calendar.current.dateComponents([.year, .month, .day], from: Date())
        dateComponents.hour = time.hour
        dateComponents.minute = time.minute
        
        let trigger = UNCalendarNotificationTrigger(dateMatching: dateComponents, repeats: true)
        let request = UNNotificationRequest(identifier: "dailyReminder", content: content, trigger: trigger)
        
        try await center.add(request)
    }
    
    func cancelReminder() {
        let center = UNUserNotificationCenter.current()
        center.removePendingNotificationRequests(withIdentifiers: ["dailyReminder"])
    }
}
```

```swift
enum DemoDataService {
    @MainActor static func seed(context: ModelContext, clock: Clock) {
        let demoDeck = Deck(id: UUID(), name: "Demo Deck", createdAt: clock.now(), isArchived: false, isDemo: true)
        
        let demoCards = [
            Card(id: UUID(), front: "Question 1", back: "Answer 1", note: nil, deck: demoDeck, easeFactor: 2.5, intervalDays: 1, repetitions: 0, nextReviewDate: clock.now().addingTimeInterval(86400), isDemo: true),
            Card(id: UUID(), front: "Question 2", back: "Answer 2", note: nil, deck: demoDeck, easeFactor: 2.5, intervalDays: 1, repetitions: 0, nextReviewDate: clock.now().addingTimeInterval(172800), isDemo: true)
        ]
        
        context.insert(demoDeck)
        context.insert(contentsOf: demoCards)
    }
    
    @MainActor static func purge(context: ModelContext) {
        let fetchRequest = ReviewLogEntry.fetchRequest()
        let results = try! context.fetch(fetchRequest)
        context.delete(contentsOf: results)
        
        for card in context.query(for: Card.self).filter(\.isDemo) {
            context.delete(card)
        }
        
        for deck in context.query(for: Deck.self).filter(\.isDemo) {
            context.delete(deck)
        }
    }
}
```
